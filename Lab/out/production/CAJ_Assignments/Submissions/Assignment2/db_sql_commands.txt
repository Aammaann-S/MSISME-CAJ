-- CAJ Lab Examination Database Setup

mysql -u root -p

--------------- CREATION ---------------

mysql> create database collegeDB;
Query OK, 1 row affected (0.00 sec)

mysql> use collegeDB;
Database changed
mysql> create table Student(Student_ID int PRIMARY KEY, Student_Name varchar(50), Program varchar(50), Semester int, Email varchar(50) UNIQUE);
Query OK, 0 rows affected (0.05 sec)

mysql> create table Subject(Subject_ID int PRIMARY KEY, Subject_Name varchar(50), Credits int, Semester int);
Query OK, 0 rows affected (0.04 sec)

mysql> create table Examination(Exam_ID int PRIMARY KEY, Subject_ID int, Exam_Date Date, Max_Marks int, FOREIGN KEY(Subject_ID) REFERENCES Subject(Subject_ID));
Query OK, 0 rows affected (0.04 sec)

mysql> create table Result(Student_ID int, Exam_ID int, Marks_Obtained int, Grade varchar(5), FOREIGN KEY(Student_ID) REFERENCES Student(Student_ID), FOREIGN KEY(Exam_ID) REFERENCES Examination(Exam_ID));
Query OK, 0 rows affected (0.05 sec)
